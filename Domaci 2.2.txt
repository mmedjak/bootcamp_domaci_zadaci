-- 1. Pronađite sve utakmice čiji naslov (`match_report`) sadrži reč `goal`, bez obzira na velika i mala slova.

SELECT *
FROM bootcamp.northern_star.matches_flat
WHERE match_report ILIKE '%goal%'
; 

-- 41


-- 2. Napravite kolonu `referee_name` tako da sadrži ime sudije ako postoji, a tekst `Unknown` ako nedostaje podatak. Prikažite 20 utakmica.


SELECT
home_team_code, home_goals, away_goals, away_team_code, 
coalesce (referee_name,'Unknown') AS referee_name
FROM bootcamp.northern_star.matches_flat
LIMIT 20
;

SELECT*
FROM bootcamp.northern_star.matches_flat
WHERE referee_name IS NULL
;

-- S obzirom da nema nedostajućih podataka za referee_name, prvi upit za sve redove vraća vrednost iz kolone referee_name.
-- Proverila sam pristup na primeru druge kolone u kojoj ima NULL:

SELECT
weather, home_team, away_team,
coalesce (weather, 'nepoznato') AS weather
FROM bootcamp.northern_star.matches_flat
;

-- 3. Prikažite utakmice čiji je naslov duži od 40 znakova. Prikažite naslov i broj znakova u njemu.

SELECT
home_team, home_goals, away_goals, away_team,
match_report,
length (match_report) AS report_characters
FROM bootcamp.northern_star.matches_flat
WHERE length (match_report) > 40
ORDER BY length (match_report) DESC
;
-- (537)

-- 4. Prikažite utakmice odigrane od 1. do 15. januara 2025, uključujući oba datuma.

SELECT *
FROM bootcamp.northern_star.matches_flat
WHERE kickoff_ts >= '2025-01-01' 
AND kickoff_ts < '2025-01-16'
;

-- 5. Prebrojte utakmice odigrane petkom. Prikažite datum i dan u nedelji za svaku od njih.

SELECT
home_team, home_goals, away_goals, away_team,
kickoff_ts,
date (kickoff_ts) AS date,
dayofweek (kickoff_ts) AS day_of_week
FROM bootcamp.northern_star.matches_flat
WHERE dayofweek (kickoff_ts) = 6
;

-- 0
-- Logika je proverena sa uslovima WHERE dayofweek (kickoff_ts) = 1 ili 7, pa upit deluje dobro, ali nema utakmica koje su igrane petkom.


-- 6. Koliko utakmica sezone `2024/25` je imalo između 20.000 i 35.000 gledalaca, uključujući obe granice?

SELECT *
FROM bootcamp.northern_star.matches_flat
WHERE season = '2024/25' 
AND attendance >=20000
AND attendance <= 35000
-- OR attendance IS NULL
ORDER BY attendance
;
--188, ovakav upit neprimetno preskače utakmice za koje nije prijavljena poseta (NULL vrednosti), ako se i takvi redovi uključe (OR attendance IS NULL), onda je 205.

-- 7. Pronađite strelce prvog gola čije ime počinje slovom `A` ili sadrži crticu. Prikažite samo različita imena.

SELECT
DISTINCT first_goal_scorer
FROM bootcamp.northern_star.matches_flat
WHERE first_goal_scorer LIKE 'A%'
OR first_goal_scorer LIKE '%-%'
;
--22

-- 8. Prikažite utakmice sezone `2024/25` koje su počele između 18:00 i 21:00, uključujući obe granice. Prikažite vreme početka i nazive timova.

SELECT
home_team, away_team,
time (kickoff_ts) AS time
FROM bootcamp.northern_star.matches_flat
WHERE season = '2024/25'
AND hour (kickoff_ts) >= 18
AND hour (kickoff_ts) <= 21
;


-- 9. Prebrojte utakmice odigrane u martu, aprilu ili maju 2025. godine čiji naslov sadrži reč `win`, bez obzira na veličinu slova.

SELECT count (*)
FROM bootcamp.northern_star.matches_flat
WHERE kickoff_ts >='2025-03-01'
AND kickoff_ts <'2025-06-01'
AND match_report ILIKE '%win%'
;
-- 5
