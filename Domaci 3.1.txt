-- 1. Za svaku sezonu prikažite ukupan broj utakmica, ukupan broj postignutih golova i prosečan broj golova po utakmici.

SELECT 
season,
count (*) AS matches,
sum (home_goals + away_goals) AS total_goals,
round (avg (home_goals + away_goals), 2) AS average_goals
FROM matches_flat
WHERE status = 'played'
GROUP BY season
;



-- 2. Za svaki stadion prikažite najmanju, najveću i prosečnu zabeleženu posetu, kao i broj utakmica za koje je poseta poznata.


SELECT
stadium_name,
min (attendance) AS minimum_attendance,
max (attendance) AS maximum_attendance,
round (avg (attendance), 2) AS average_attendance,
sum (
    CASE WHEN attendance IS NOT NULL THEN 1
    ELSE 0 END
) AS attendance_known
FROM matches_flat
GROUP BY stadium_name
ORDER BY attendance_known
;

-- 3. Prikažite sudije koje su sudile najmanje 10 utakmica i koje su ukupno pokazale najmanje 20 žutih kartona. Prikažite ime sudije, broj utakmica i ukupan broj žutih kartona.

SELECT
referee_name,
count (*) AS matches,
sum (home_yellow_cards + away_yellow_cards) AS yellow_cards
FROM matches_flat
GROUP BY referee_name
HAVING matches >= 10 
    AND yellow_cards >= 20
ORDER BY yellow_cards DESC
;


-- 4. Podelite utakmice prema ukupnom broju golova na kategorije Bez golova (0), Malo golova (1 ili 2) i Mnogo golova (3 ili više). Prikažite broj utakmica u svakoj kategoriji i najveći broj golova na utakmici u toj kategoriji.

SELECT
    count (*) Utakmice,
    max (home_goals + away_goals) as Najveci_broj_golova,
    CASE 
        WHEN home_goals + away_goals = 0 THEN 'Bez golova'
        WHEN home_goals + away_goals = 1 OR home_goals + away_goals = 2 THEN  'Malo golova'
        ELSE 'Mnogo golova'
        END as Kategorija 

FROM matches_flat
GROUP BY Kategorija
;

-- 5. Za svaki klub prikažite broj utakmica koje je odigrao kao domaćin, broj pobeda kod kuće i broj poraza kod kuće. U rezultat uključite samo klubove koji su kod kuće odigrali najmanje 10 utakmica.

SELECT 
    home_team,
    count (*) AS home_matches,
    sum (
        CASE 
            WHEN home_goals > away_goals THEN 1
            ELSE 0
        END) as home_wins,
    sum (
        CASE 
            WHEN home_goals < away_goals THEN 1
            ELSE 0
        END) AS home_losses
FROM matches_flat
GROUP BY home_team
HAVING home_matches >= 10
ORDER BY home_matches DESC, home_wins DESC
;

-- 6. Za svakog sudiju prikažite ukupan broj utakmica, broj utakmica u kojima je pokazao bar jedan crveni karton i procenat takvih utakmica u odnosu na sve utakmice koje je sudio.  

SELECT
referee_name,
count (*) AS all_matches,
sum (
    CASE 
        WHEN home_red_cards + away_red_cards >= 1 THEN 1
        ELSE 0 
    END
    ) AS red_card_matches,
round ((red_card_matches / all_matches * 100),2) AS red_card_matches_percentage
FROM matches_flat
GROUP BY referee_name
;


-- 7. Za svaku sezonu prikažite broj pobeda domaćina, broj nerešenih utakmica i broj pobeda gostiju. Takođe prikažite procenat utakmica koje su završene nerešenim rezultatom.

SELECT
season,
count (*) AS all_matches,
sum (
    CASE WHEN home_goals > away_goals THEN 1 
    ELSE 0
    END
) AS home_wins,
sum (
    CASE WHEN home_goals = away_goals THEN 1 
    ELSE 0
    END
) AS ties,
sum (
    CASE WHEN home_goals < away_goals THEN 1 
    ELSE 0
    END
) AS away_wins,
round (ties/all_matches*100 , 2) AS ties_percentage
FROM matches_flat
GROUP BY season
;
