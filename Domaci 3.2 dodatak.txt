-- Osnovni nivo:
-- 1.Za svakog sudiju prikažite njegovo ime i broj utakmica koje je sudio. Uključite i sudije koje nisu sudile ni jednu utakmicu.

SELECT
    r.referee_id,
    r.referee_name
FROM referees r
LEFT JOIN matches m ON r.referee_id = m.referee_id
WHERE m.match_id IS NULL;

SELECT 
    r.referee_id, r.referee_name,
    count (m.match_id)        
FROM referees r
LEFT JOIN matches m ON r.referee_id = m.referee_id
GROUP BY r.referee_id, r.referee_name
;



-- 2.Prikažite imena i prezimena igrača koji nemaju ni jedan zapis u tabeli player_match_stats.

SELECT
    p.first_name, p.last_name
FROM players p
LEFT JOIN player_match_stats s ON p.player_id = s.player_id
WHERE match_id IS NULL; 

-- 3.Prikažite broj golova koji je svaki klub postigao kao domaćin u sezoni 2024/25. Rezultat sortirajte od najvećeg ka najmanjem broju golova.

SELECT 
    count (*) goals,
    e.team_id,
    t.team_name
FROM matches m
INNER JOIN match_events e ON m.match_id = e.match_id
INNER JOIN teams t ON e.team_id = t.team_id 
WHERE e.event_type = 'goal'
    AND m.season = '2024/25'
GROUP BY e.team_id, t.team_name
ORDER BY goals DESC
;


-- Srednji nivo:
-- 1.Za svaku sezonu prikažite broj golova koje su postigli igrači na pozicijama FW, MF i DF. Rezultat grupišite prema sezoni i poziciji igrača.

SELECT
 count (*) goals,
 p.position,
 m.season
FROM match_events e
INNER JOIN players p ON e.player_id = p.player_id 
INNER JOIN matches m ON e.match_id = m.match_id
WHERE e.event_type = 'goal'
AND p.position IN ('FW', 'MF', 'DF')
GROUP BY m.season, p.position
ORDER BY m.season 
;

-- 2.Za svaki stadion prikažite njegov naziv i ukupan broj gledalaca na utakmicama odigranim na njemu. Uključite i stadione za koje nema evidentiranih utakmica. 



SELECT
    s.stadium_name, 
    sum (m.attendance) total_attendance
FROM stadiums s
LEFT JOIN matches m ON s.stadium_id = m.stadium_id
GROUP BY s.stadium_name
ORDER BY total_attendance DESC
;


-- 3.Za svakog sudiju prikažite broj različitih utakmica u kojima je dosuđen penal, bez obzira na to da li je penal realizovan. Uključite i sudije za koje nema takvih utakmica.

SELECT DISTINCT event_type 
FROM match_events; --penalty_miss, penalty_goal


SELECT
    count (distinct e.match_id) penalty_matches, 
    r.referee_id, r.referee_name
FROM referees r 
LEFT JOIN matches m on r.referee_id = m.referee_id
LEFT JOIN match_events e ON m.match_id = e.match_id
    AND e.event_type IN ('penalty_miss', 'penalty_goal')
GROUP BY r.referee_id, r.referee_name
ORDER BY penalty_matches DESC;
